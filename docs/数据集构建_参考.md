## 一、测试集核心设计思路

### 1. 评估维度对齐你的模板

你的调度器输出包含三个层次，测试集也应分层评估：

| 层次 | 输出内容 | 评估方式 |
|------|----------|----------|
| **L1 意图识别** | HRI=是/否 | 二元判断（精确匹配） |
| **L2 阶段划分** | 阶段标签序列 | 序列相似度（编辑距离/Levenshtein） |
| **L3 动作排序** | 原子动作序列 | **排序题匹配**（核心） |

### 2. "排序题"评分机制

将模型的动作序列输出与标准答案进行**拓扑序一致性**比对，而非逐字相等：

## 三、自动评分引擎（Python）

以下代码可直接集成到你的测试流程中，支持**排序正确性判断**：

```python
#!/usr/bin/env python3
"""
任务规划测试集评分引擎
支持：HRI判断、阶段划分、动作序列拓扑排序评分
"""

import json
from typing import List, Dict, Tuple
from dataclasses import dataclass
import re


@dataclass
class ActionStep:
    id: int
    action: str
    target: str
    phase: str
    deps: List[int]


class TaskPlanEvaluator:
    def __init__(self, gold_standard: Dict):
        self.gold = gold_standard
        self.gold_actions = {a["id"]: ActionStep(**a) for a in gold_standard["actions_gold"]}
        self.gold_stages = gold_standard["stages_expected"]
        self.gold_hri = gold_standard["hri_expected"]
        
        # 同义词映射（从 SKILL_LIB 提取）
        self.synonyms = {
            "PICK": "GRASP",
            "TAKE": "GRASP",
            "REPORT": "SPEAK",
            "TELL": "SPEAK",
            "GIVE": "HANDOVER",
            "DELIVER": "HANDOVER",
        }

    def parse_model_output(self, raw_text: str) -> Dict:
        """
        解析模型文本输出，提取结构化信息
        适配你的 SCHEDULER 模板输出格式
        """
        result = {
            "hri": None,
            "stages": [],
            "actions": []
        }
        
        # 1. 提取 HRI 判断
        hri_match = re.search(r'HRI[：:]\s*(是|否|yes|no|true|false)', raw_text, re.I)
        if hri_match:
            val = hri_match.group(1).lower()
            result["hri"] = val in ("是", "yes", "true")
        
        # 2. 提取阶段划分
        stage_match = re.search(r'阶段[：:]\s*([^\n]+)', raw_text)
        if stage_match:
            stages_text = stage_match.group(1)
            # 支持 "备料→烹饪→组装→递送" 或 "备料、烹饪、组装、递送"
            result["stages"] = re.split(r'[→、,;，\s]+', stages_text.strip())
        
        # 3. 提取动作序列（适配你的格式：序号. 动作|目标|目的）
        action_pattern = re.compile(
            r'(\d+)\.\s*([A-Z_]+)\|([^|]+)(?:\|([^|\n]*))?',
            re.MULTILINE
        )
        for match in action_pattern.finditer(raw_text):
            step_num = int(match.group(1))
            action = match.group(2).strip().upper()
            target = match.group(3).strip()
            
            # 同义词归一化
            action = self.synonyms.get(action, action)
            
            result["actions"].append({
                "step": step_num,
                "action": action,
                "target": target
            })
        
        return result

    def score_hri(self, pred_hri: bool) -> Tuple[float, str]:
        """HRI 判断评分：精确匹配"""
        if pred_hri is None:
            return 0.0, "未提取到HRI判断"
        correct = (pred_hri == self.gold_hri)
        return (1.0 if correct else 0.0, "正确" if correct else f"错误（应为{'是' if self.gold_hri else '否'}）")

    def score_stages(self, pred_stages: List[str]) -> Tuple[float, str]:
        """阶段划分评分：序列相似度"""
        if not pred_stages:
            return 0.0, "未提取到阶段"
        
        # 计算最长公共子序列（LCS）比例
        lcs_len = self._lcs_length(self.gold_stages, pred_stages)
        score = lcs_len / max(len(self.gold_stages), len(pred_stages))
        
        # 额外惩罚：阶段顺序错误
        if pred_stages != self.gold_stages[:len(pred_stages)]:
            score *= 0.8
            
        return round(score, 2), f"预测: {pred_stages} | 标准: {self.gold_stages}"

    def score_actions(self, pred_actions: List[Dict]) -> Tuple[float, str]:
        """
        动作序列评分（核心：排序题逻辑）
        
        评分维度：
        1. 动作覆盖度（是否遗漏关键动作）
        2. 依赖满足度（前置动作是否在之前）
        3. 动作正确性（动作类型是否匹配）
        """
        if not pred_actions:
            return 0.0, "未提取到动作序列"
        
        total_score = 0.0
        details = []
        
        # 构建 gold 的依赖图
        gold_deps = {aid: set(self.gold_actions[aid].deps) for aid in self.gold_actions}
        
        # 1. 检查动作覆盖（召回率）
        gold_action_set = {a.action for a in self.gold_actions.values()}
        pred_action_set = {a["action"] for a in pred_actions}
        coverage = len(gold_action_set & pred_action_set) / len(gold_action_set)
        total_score += coverage * 0.3
        details.append(f"动作覆盖: {coverage:.2f}")
        
        # 2. 检查依赖满足度（拓扑排序正确性）
        # 将预测序列映射到 gold id（通过动作+目标模糊匹配）
        pred_to_gold = self._map_pred_to_gold(pred_actions)
        dep_violations = 0
        seen_gold_ids = set()
        
        for pred_step in pred_actions:
            mapped_gold_id = pred_to_gold.get(pred_step["step"])
            if mapped_gold_id:
                # 检查该动作的所有依赖是否已出现
                for dep_id in gold_deps.get(mapped_gold_id, set()):
                    if dep_id not in seen_gold_ids:
                        dep_violations += 1
                seen_gold_ids.add(mapped_gold_id)
        
        dep_score = max(0, 1.0 - (dep_violations / max(len(self.gold_actions), 1)))
        total_score += dep_score * 0.4
        details.append(f"依赖满足: {dep_score:.2f} (违规{dep_violations}次)")
        
        # 3. 检查阶段归属正确性
        phase_correct = 0
        for pred_step in pred_actions:
            mapped_gold_id = pred_to_gold.get(pred_step["step"])
            if mapped_gold_id:
                gold_phase = self.gold_actions[mapped_gold_id].phase
                # 这里简化处理：假设阶段信息可以从输出中提取
                # 实际可扩展为检查动作是否出现在正确阶段标题下
                phase_correct += 1
        
        phase_score = phase_correct / max(len(pred_actions), 1)
        total_score += phase_score * 0.3
        details.append(f"动作识别: {phase_score:.2f}")
        
        return round(min(total_score, 1.0), 2), " | ".join(details)

    def _map_pred_to_gold(self, pred_actions: List[Dict]) -> Dict[int, int]:
        """将预测步骤映射到标准步骤（基于动作+目标模糊匹配）"""
        mapping = {}
        used_gold = set()
        
        for pred in pred_actions:
            best_match = None
            best_score = -1
            
            for gid, gold in self.gold_actions.items():
                if gid in used_gold:
                    continue
                
                # 动作匹配
                act_match = 1.0 if pred["action"] == gold.action else 0.0
                # 目标模糊匹配（包含关系或相等）
                tgt_match = 1.0 if (pred["target"] in gold.target or gold.target in pred["target"]) else 0.0
                
                score = act_match * 0.6 + tgt_match * 0.4
                if score > best_score and score > 0.5:
                    best_score = score
                    best_match = gid
            
            if best_match:
                mapping[pred["step"]] = best_match
                used_gold.add(best_match)
        
        return mapping

    def _lcs_length(self, seq1: List[str], seq2: List[str]) -> int:
        """计算最长公共子序列长度"""
        m, n = len(seq1), len(seq2)
        dp = [[0] * (n + 1) for _ in range(m + 1)]
        
        for i in range(1, m + 1):
            for j in range(1, n + 1):
                if seq1[i-1] == seq2[j-1]:
                    dp[i][j] = dp[i-1][j-1] + 1
                else:
                    dp[i][j] = max(dp[i-1][j], dp[i][j-1])
        
        return dp[m][n]

    def evaluate(self, model_raw_output: str) -> Dict:
        """完整评估入口"""
        parsed = self.parse_model_output(model_raw_output)
        
        hri_score, hri_detail = self.score_hri(parsed["hri"])
        stage_score, stage_detail = self.score_stages(parsed["stages"])
        action_score, action_detail = self.score_actions(parsed["actions"])
        
        weights = self.gold.get("scoring_rules", {})
        w_hri = weights.get("hri_weight", 0.15)
        w_stage = weights.get("stage_weight", 0.25)
        w_action = weights.get("action_weight", 0.60)
        
        final = hri_score * w_hri + stage_score * w_stage + action_score * w_action
        
        return {
            "total_score": round(final, 3),
            "breakdown": {
                "hri": {"score": hri_score, "detail": hri_detail},
                "stages": {"score": stage_score, "detail": stage_detail},
                "actions": {"score": action_score, "detail": action_detail}
            },
            "parsed": parsed
        }


# ==================== 使用示例 ====================

if __name__ == "__main__":
    # 加载测试集
    with open("task_plan_testset.json", "r", encoding="utf-8") as f:
        testset = json.load(f)
    
    # 模拟模型输出（实际使用时替换为 API 返回）
    model_output = """
    HRI：是 | 需靠近用户+递交+告知
    阶段：备料→烹饪→组装→递送
    
    1. NAVIGATE|冰箱|取鸡肉
    2. OPEN|冰箱门|打开
    3. GRASP|鸡肉|取出
    4. WASH|鸡肉|清洗
    5. CUT|鸡肉|切块
    6. NAVIGATE|灶台|去灶台
    7. HEAT|油锅|热油
    8. COOK|鸡排|炸金黄
    9. NAVIGATE|台面|去台面
    10. GRASP|面包|取
    11. PLACE|鸡排放面包|组装
    12. NAVIGATE|用户位置|靠近
    13. SPEAK|告知|说您的咖啡
    14. HANDOVER|汉堡|递上
    收尾：BOW|鞠躬|致谢
    """
    
    evaluator = TaskPlanEvaluator(testset[0])
    result = evaluator.evaluate(model_output)
    
    print(f"总分: {result['total_score']}")
    print(json.dumps(result["breakdown"], ensure_ascii=False, indent=2))
```

## 六、排序题变体（可选扩展）

如果你想更严格地以"排序题"形式测试（类似考试中的拖拽排序），可以设计为：

```json
{
  "test_id": "SORT-001",
  "type": "sorting",
  "instruction": "把以下动作按正确顺序排列：递水给客人",
  "scrambled_actions": [
    "HANDOVER|水杯",
    "NAVIGATE|客人位置", 
    "GRASP|水杯",
    "POUR|水",
    "SPEAK|请用水"
  ],
  "correct_order": [2, 3, 4, 5, 1]  // NAVIGATE → GRASP → POUR → SPEAK → HANDOVER
}
```

评分时直接比对模型输出的序号序列与 `correct_order`，计算**Kendall Tau 距离**或**Spearman 相关系数**：

```python
from scipy.stats import kendalltau

def score_sorting(pred_order, correct_order):
    # Kendall Tau: 1.0 表示完全一致，-1.0 表示完全相反
    tau, _ = kendalltau(pred_order, correct_order)
    # 映射到 0-1 分数
    return (tau + 1) / 2
```
